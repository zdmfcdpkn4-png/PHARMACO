==============================================================================
 PHARMACO — Installation complete sur Raspberry Pi (API + PostgreSQL + Frontend)
 Tout tourne sur le Pi. Une seule methode, etape par etape, a copier-coller.
==============================================================================

PRE-REQUIS
- Raspberry Pi 4 ou 5, Raspberry Pi OS 64-bit (Bookworm), connecte au reseau.
- Acces au terminal (clavier+ecran, ou SSH).
- Un compte avec sudo (les commandes supposent l'utilisateur "pi" ; si ton
  utilisateur est different, remplace "pi" partout par ton nom d'utilisateur).

Note : execute les blocs dans l'ordre. Les lignes commencant par "#" sont des
commentaires, tu peux les coller aussi.

------------------------------------------------------------------------------
ETAPE 1 — Mettre a jour le systeme et installer les paquets
------------------------------------------------------------------------------
sudo apt update && sudo apt full-upgrade -y
sudo apt install -y curl ca-certificates gnupg lsb-release git

# --- PostgreSQL 16 (depot officiel PGDG) ---
sudo install -d /usr/share/postgresql-common/pgdg
curl -fsSL https://www.postgresql.org/media/keys/ACCC4CF8.asc | sudo gpg --dearmor -o /usr/share/postgresql-common/pgdg/apt.postgresql.org.gpg
echo "deb [signed-by=/usr/share/postgresql-common/pgdg/apt.postgresql.org.gpg] http://apt.postgresql.org/pub/repos/apt $(lsb_release -cs)-pgdg main" | sudo tee /etc/apt/sources.list.d/pgdg.list

# --- Node.js 20 (depot NodeSource) ---
curl -fsSL https://deb.nodesource.com/setup_20.x | sudo -E bash -

# --- Caddy (serveur web + reverse proxy) ---
sudo apt install -y debian-keyring debian-archive-keyring apt-transport-https
curl -1sLf 'https://dl.cloudsmith.io/public/caddy/stable/gpg.key' | sudo gpg --dearmor -o /usr/share/keyrings/caddy-stable-archive-keyring.gpg
curl -1sLf 'https://dl.cloudsmith.io/public/caddy/stable/debian.deb.txt' | sudo tee /etc/apt/sources.list.d/caddy-stable.list

# --- Installer le tout ---
sudo apt update
sudo apt install -y postgresql-16 postgresql-client-16 nodejs caddy

# Verifications
node -v          # doit afficher v20.x
psql --version   # doit afficher 16.x
sudo systemctl enable --now postgresql

------------------------------------------------------------------------------
ETAPE 2 — Creer la base de donnees et son utilisateur
------------------------------------------------------------------------------
# Remplace 'MotDePasseSolide' par un mot de passe a toi (sans @ : / # ? pour
# simplifier). Note-le, il servira a l'etape 4.

sudo -u postgres psql <<'SQL'
CREATE USER pharmaco WITH PASSWORD 'MotDePasseSolide';
CREATE DATABASE pharmaco OWNER pharmaco;
GRANT ALL PRIVILEGES ON DATABASE pharmaco TO pharmaco;
SQL

# Test : doit afficher "1"
psql "postgres://pharmaco:MotDePasseSolide@localhost:5432/pharmaco" -c "SELECT 1;"

------------------------------------------------------------------------------
ETAPE 3 — Recuperer le code de l'application
------------------------------------------------------------------------------
cd ~
git clone https://github.com/zdmfcdpkn4-png/PHARMACO.git
cd PHARMACO

------------------------------------------------------------------------------
ETAPE 4 — Importer tes donnees actuelles depuis Render
------------------------------------------------------------------------------
# But : transferer la base hebergee sur Render vers le Pi.
# 1) Sur https://dashboard.render.com -> base "pharmaco-db" -> onglet "Connect"
#    -> copie la "External Database URL".
# 2) Colle-la ci-dessous (entre les guillemets) en ajoutant ?sslmode=require :

export RENDER_URL="postgres://UTILISATEUR:MOTDEPASSE@HOTE.render.com/NOMBASE?sslmode=require"

# 3) Exporter depuis Render, puis importer dans le Pi :
pg_dump "$RENDER_URL" -Fc --no-owner --no-privileges -f ~/pharmaco.dump
pg_restore --no-owner --no-privileges -d "postgres://pharmaco:MotDePasseSolide@localhost:5432/pharmaco" ~/pharmaco.dump

# Verifier que les donnees sont la :
psql "postgres://pharmaco:MotDePasseSolide@localhost:5432/pharmaco" -c "SELECT id, name FROM boards;"

------------------------------------------------------------------------------
ETAPE 5 — Configurer et lancer l'API (backend)
------------------------------------------------------------------------------
cd ~/PHARMACO/server
npm install

# Genere un secret d'authentification (copie le resultat affiche) :
node -e "console.log(require('crypto').randomBytes(48).toString('hex'))"

# Cree le fichier .env (remplace MotDePasseSolide et colle le SECRET genere) :
cat > .env <<'ENV'
DATABASE_URL=postgres://pharmaco:MotDePasseSolide@localhost:5432/pharmaco
PGSSL=false
AUTH_SECRET=COLLE_ICI_LE_SECRET_GENERE
CLIENT_ORIGIN=*
PORT=4000
RUN_MIGRATIONS=true
ENV

# Installer l'API comme service qui demarre tout seul :
sudo tee /etc/systemd/system/pharmaco-api.service >/dev/null <<'UNIT'
[Unit]
Description=PHARMACO API
After=network.target postgresql.service
Wants=postgresql.service
[Service]
Type=simple
User=pi
WorkingDirectory=/home/pi/PHARMACO/server
EnvironmentFile=/home/pi/PHARMACO/server/.env
ExecStart=/usr/bin/node src/index.js
Restart=always
RestartSec=3
[Install]
WantedBy=multi-user.target
UNIT

sudo systemctl daemon-reload
sudo systemctl enable --now pharmaco-api

# Verifier (doit repondre {"status":"ok",...}) :
sleep 2
curl -s http://localhost:4000/api/health ; echo

------------------------------------------------------------------------------
ETAPE 6 — Construire le frontend et le publier
------------------------------------------------------------------------------
cd ~/PHARMACO/client
npm install
VITE_USE_MOCK=false VITE_API_URL=/api npm run build

# Copier les fichiers construits la ou Caddy les servira :
sudo mkdir -p /var/www/pharmaco
sudo rm -rf /var/www/pharmaco/*
sudo cp -r dist/* /var/www/pharmaco/

------------------------------------------------------------------------------
ETAPE 7 — Configurer Caddy (sert le site + relaie /api vers l'API)
------------------------------------------------------------------------------
sudo tee /etc/caddy/Caddyfile >/dev/null <<'CADDY'
:80 {
    encode gzip
    handle /api/* {
        reverse_proxy localhost:4000
    }
    handle {
        root * /var/www/pharmaco
        try_files {path} /index.html
        file_server
    }
}
CADDY

sudo systemctl reload caddy

ETAPE 9 — Sauvegarde automatique de la base (tous les jours a 3h)
------------------------------------------------------------------------------
mkdir -p ~/backups
( crontab -l 2>/dev/null ; echo '0 3 * * * pg_dump "postgres://pharmaco:MotDePasseSolide@localhost:5432/pharmaco" -Fc -f ~/backups/pharmaco-$(date +\%F).dump && find ~/backups -name "*.dump" -mtime +14 -delete' ) | crontab -

# Restaurer une sauvegarde si besoin :
#   pg_restore --no-owner -d "postgres://pharmaco:MotDePasseSolide@localhost:5432/pharmaco" ~/backups/pharmaco-AAAA-MM-JJ.dump

COMMANDES UTILES
- Etat / logs de l'API :   sudo systemctl status pharmaco-api
                           journalctl -u pharmaco-api -f
- Redemarrer l'API :       sudo systemctl restart pharmaco-api
- Recharger Caddy :        sudo systemctl reload caddy
- Etat PostgreSQL :        pg_lsclusters

METTRE A JOUR L'APPLICATION PLUS TARD
  cd ~/PHARMACO && git pull
  cd server && npm install && sudo systemctl restart pharmaco-api
  cd ../client && npm install && VITE_USE_MOCK=false VITE_API_URL=/api npm run build
  sudo rm -rf /var/www/pharmaco/* && sudo cp -r dist/* /var/www/pharmaco/

DEPANNAGE
- "column color does not exist" : verifie RUN_MIGRATIONS=true dans server/.env
  puis  sudo systemctl restart pharmaco-api
- Page inaccessible : verifie  sudo systemctl status caddy  et l'IP du Pi.
- API qui ne demarre pas : journalctl -u pharmaco-api -n 50
- "peer authentication failed" : garde bien "localhost" dans DATABASE_URL.
